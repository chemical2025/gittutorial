from pyspark.sql import SparkSession
import json
from google.cloud import storage

# Initialize Spark session
spark = SparkSession.builder \
    .appName("SQLServerToGCS") \
    .config("spark.driver.extraClassPath", "/path/to/sqljdbc42.jar") \
    .getOrCreate()

# GCS bucket and config file path
bucket_name = "my-etl-bucket"
config_blob_path = "config/config.json"
output_path = "gs://my-etl-bucket/output/"

# Initialize GCS client
client = storage.Client()
bucket = client.get_bucket(bucket_name)
blob = bucket.blob(config_blob_path)
config_data = json.loads(blob.download_as_text())

# Extract config values
sql_config = config_data["sql_server"]
last_processed = config_data["last_processed"]

# JDBC connection
jdbc_url = f"jdbc:sqlserver://{sql_config['host']}:{sql_config['port']};databaseName={sql_config['database']}"
connection_properties = {
    "user": sql_config["username"],
    "password": sql_config["password"],
    "driver": "com.microsoft.sqlserver.jdbc.SQLServerDriver"
}

# SQL query to fetch new records
query = f"(SELECT * FROM {sql_config['table']} WHERE {last_processed['column']} > '{last_processed['value']}') AS subquery"
df = spark.read.jdbc(url=jdbc_url, table=query, properties=connection_properties)

# Write to GCS in Parquet format
if df.count() > 0:
    df.write.mode("overwrite").parquet(output_path)

    # Update last_processed timestamp
    latest_value = df.agg({last_processed['column']: "max"}).collect()[0][0]
    config_data["last_processed"]["value"] = str(latest_value)
    blob.upload_from_string(json.dumps(config_data, indent=4), content_type="application/json")

print("PySpark job completed successfully.")
